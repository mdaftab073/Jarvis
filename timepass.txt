            document.addEventListener('click', function (evt) {
                try {
                    var btn = document.getElementById('btnSubmit');
                    if (!btn || !(evt.target === btn || btn.contains(evt.target))) {
                        return;
                    }
                    var pwdField = document.getElementById('txt_password');
                    //Modified on 26/09/2026 - Browser "Save Password" fix: ciphertext goes to hdnEncPwd, not into the password box
                    var encField = document.getElementById('hdnEncPwd');
                    var modulus = document.getElementById('hdnRsaModulus').value;
                    var exponent = document.getElementById('hdnRsaExponent').value;
                    encField.value = '';
                    if (pwdField && pwdField.value && modulus && exponent && window.JSEncrypt) {
                        var encrypt = new JSEncrypt();
                        encrypt.getKey().setPublic(modulus, exponent);
                        var cipher = encrypt.encrypt(pwdField.value);
                        //Old code: if (cipher) { pwdField.value = cipher; }
                        if (cipher) {
                            // Send the ciphertext in a separate hidden field and leave the visible
                            // password box untouched, so the browser's "Save password" stores the
                            // real password (not ciphertext that would be re-encrypted on autofill).
                            // Removing the box's name keeps the plain password out of the request body.
                            encField.value = cipher;
                            pwdField.removeAttribute('name');
                        } else {
                            // Encryption fails when the box holds an old saved ciphertext (too long for RSA),
                            // left in the browser's password manager by the previous version of this page.
                            evt.preventDefault();
                            evt.stopPropagation();
                            pwdField.value = '';
                            pwdField.focus();
                            alert('Saved password is not valid. Please type your password again.');
                        }
                        //End Modified on 26/09/2026
                    }
                } catch (ex) {
                    //Encryption failed - fall through and submit as-is; server rejects a non-ciphertext
                    //password rather than accepting it in plain text (see ResolveLoginPassword)
                }
            }, true);
        </script>
